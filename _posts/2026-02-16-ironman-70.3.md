function niceMax(v){
  if (v <= 0) return 10;
  const mag = Math.pow(10, Math.floor(Math.log10(v)));
  const steps = [1, 2, 2.5, 5, 10];
  for (const s of steps){
    if (v <= s * mag) return s * mag;
  }
  return 10 * mag;
}

let isolatedSeries = null; // null = show all lines; otherwise a SERIES_DEFS key

function renderStatsChart(){
  const svg = document.getElementById('statsSvg');
  svg.innerHTML = '';

  const n = WEEKLY_STATS.length;
  const VBW = 900, VBH = 340;
  const marginLeft = 42, marginRight = 42, marginTop = 16, marginBottom = 46;
  const plotW = VBW - marginLeft - marginRight;
  const plotH = VBH - marginTop - marginBottom;
  const stepX = n > 1 ? plotW / (n - 1) : 0;
  const xAt = i => marginLeft + stepX * i;

  const ns = 'http://www.w3.org/2000/svg';
  const el = (tag, attrs) => {
    const e = document.createElementNS(ns, tag);
    for (const k in attrs) e.setAttribute(k, attrs[k]);
    return e;
  };

  const activeDefs = isolatedSeries ? SERIES_DEFS.filter(s => s.key === isolatedSeries) : SERIES_DEFS;
  const isolatedDef = isolatedSeries ? SERIES_DEFS.find(s => s.key === isolatedSeries) : null;

  // When isolated, the one active line gets its own scale on the left axis
  // (in its own unit) so a small line like Swim isn't a flat streak at the
  // bottom of a 90-mile axis. When showing everything, miles go on the left
  // and hours go on the right, same as before.
  let leftMax, leftUnit, rightMax;
  if (isolatedDef){
    leftMax = niceMax(Math.max(...WEEKLY_STATS.map(w => w[isolatedDef.key])));
    leftUnit = isolatedDef.axis === 'hours' ? 'hrs' : 'mi';
    rightMax = null;
  } else {
    leftMax = niceMax(Math.max(...WEEKLY_STATS.map(w => w.totalMiles)));
    leftUnit = 'mi';
    rightMax = niceMax(Math.max(...WEEKLY_STATS.map(w => w.totalHours)));
  }

  const yLeft = v => marginTop + plotH - (v / leftMax) * plotH;
  const yRight = rightMax != null ? (v => marginTop + plotH - (v / rightMax) * plotH) : null;
  const scaleFor = def => (!isolatedDef && def.axis === 'hours') ? yRight : yLeft;

  // gridlines + axis labels (4 bands)
  for (let g = 0; g <= 4; g++){
    const frac = g / 4;
    const y = marginTop + plotH - frac * plotH;
    svg.appendChild(el('line', { class: 'gridline', x1: marginLeft, y1: y, x2: VBW - marginRight, y2: y }));
    svg.appendChild(el('text', { class: 'axis-label left', x: marginLeft - 8, y: y + 3 }));
    svg.lastChild.textContent = Math.round(leftMax * frac) + (isolatedDef ? ` ${leftUnit}` : '');
    if (rightMax != null){
      svg.appendChild(el('text', { class: 'axis-label right', x: VBW - marginRight + 8, y: y + 3 }));
      svg.lastChild.textContent = (rightMax * frac).toFixed(rightMax * frac < 10 ? 1 : 0);
    }
  }

  // week labels (rotated), every week
  WEEKLY_STATS.forEach((w, i) => {
    const x = xAt(i);
    const t = el('text', {
      class: 'week-label', x: x, y: marginTop + plotH + 16,
      transform: `rotate(-55 ${x} ${marginTop + plotH + 16})`,
      'text-anchor': 'end'
    });
    t.textContent = w.dateRange.split('–')[0];
    svg.appendChild(t);
  });

  activeDefs.forEach(s => {
    const scale = scaleFor(s);
    const pts = WEEKLY_STATS.map((w, i) => `${xAt(i)},${scale(w[s.key])}`).join(' ');
    const attrs = { class: 'series-line', points: pts, stroke: s.color, 'stroke-width': s.width };
    if (s.dash) attrs['stroke-dasharray'] = s.dash;
    svg.appendChild(el('polyline', attrs));

    WEEKLY_STATS.forEach((w, i) => {
      const c = el('circle', { class: 'pt', cx: xAt(i), cy: scale(w[s.key]), r: 2, fill: s.color });
      const title = document.createElementNS(ns, 'title');
      title.textContent = `${s.label} — ${w.dateRange}: ${w[s.key]}`;
      c.appendChild(title);
      svg.appendChild(c);
    });
  });

  // legend — click a line to isolate it, click again (or "Show all lines") to bring the rest back
  const legend = document.getElementById('statsLegend');
  legend.innerHTML = '';
  SERIES_DEFS.forEach(s => {
    const item = document.createElement('div');
    const isActive = isolatedSeries === s.key;
    item.className = 'legend-item' + (isActive ? ' active' : '') + (isolatedSeries && !isActive ? ' dimmed' : '');
    item.setAttribute('role', 'button');
    item.setAttribute('tabindex', '0');
    item.setAttribute('aria-pressed', isActive ? 'true' : 'false');
    item.setAttribute('aria-label', `Show only ${s.label}`);
    const swSvg = `<svg width="24" height="10" viewBox="0 0 24 10"><line x1="1" y1="5" x2="23" y2="5" stroke="${s.color}" stroke-width="${s.width}" ${s.dash ? `stroke-dasharray="${s.dash}"` : ''}/></svg>`;
    item.innerHTML = swSvg + `<span style="color:${s.color}">${s.label}</span>`;

    const toggle = () => {
      isolatedSeries = isolatedSeries === s.key ? null : s.key;
      renderStatsChart();
    };
    item.addEventListener('click', toggle);
    item.addEventListener('keydown', e => {
      if (e.key === 'Enter' || e.key === ' '){ e.preventDefault(); toggle(); }
    });
    legend.appendChild(item);
  });

  const resetBtn = document.getElementById('statsReset');
  resetBtn.hidden = !isolatedSeries;
}

document.getElementById('statsReset').addEventListener('click', () => {
  isolatedSeries = null;
  renderStatsChart();
});

renderStatsChart();

const svg = document.getElementById('profileSvg');
const VB_W = 900, VB_H = 150, TOP_PAD = 22, BASE_Y = 118;
const AXIS_W = 66, RIGHT_PAD = 20;
const n = WEEKS.length;
const stepX = n > 1 ? (VB_W - AXIS_W - RIGHT_PAD) / (n - 1) : 0;
const startX = AXIS_W;

function moodToY(mood){
  // mood 1..5 -> y position (higher mood = higher on chart = smaller y)
  const usable = BASE_Y - TOP_PAD;
  return BASE_Y - ((mood - 1) / 4) * usable;
}

let points = WEEKS.map((w, i) => ({
  x: startX + stepX * i,
  y: moodToY(w.mood),
  w
}));

// polyline / fill
const lineStr = points.map(p => `${p.x},${p.y}`).join(' ');
svg.querySelector('.line').setAttribute('points', lineStr);
const fillStr = `${startX},${BASE_Y} ` + lineStr + ` ${points[points.length-1].x},${BASE_Y}`;
svg.querySelector('.fill').setAttribute('points', fillStr);
svg.querySelector('.baseline').setAttribute('x1', startX);

// y-axis mood labels
const nsY = 'http://www.w3.org/2000/svg';
for (let mood = 1; mood <= 5; mood++){
  const t = document.createElementNS(nsY, 'text');
  t.setAttribute('class', 'mood-axis-label');
  t.setAttribute('x', AXIS_W - 10);
  t.setAttribute('y', moodToY(mood) + 3);
  t.setAttribute('text-anchor', 'end');
  t.textContent = MOOD_LABELS[mood];
  svg.appendChild(t);
}

let activeIndex = WEEKS.length - 1;

points.forEach((p, i) => {
  const g = document.createElementNS('http://www.w3.org/2000/svg','g');
  g.setAttribute('class', 'tab-point');
  g.setAttribute('tabindex', '0');
  g.setAttribute('role', 'tab');
  g.setAttribute('aria-label', `Week ${p.w.week}, ${MOOD_LABELS[p.w.mood]}`);
  g.dataset.index = i;

  const focusRing = document.createElementNS('http://www.w3.org/2000/svg','rect');
  focusRing.setAttribute('class','focus-ring');
  focusRing.setAttribute('x', p.x - 12);
  focusRing.setAttribute('y', p.y - 12);
  focusRing.setAttribute('width', 24);
  focusRing.setAttribute('height', 24);

  const dot = document.createElementNS('http://www.w3.org/2000/svg','circle');
  dot.setAttribute('class','dot');
  dot.setAttribute('cx', p.x);
  dot.setAttribute('cy', p.y);
  dot.setAttribute('r', 5);

  const label = document.createElementNS('http://www.w3.org/2000/svg','text');
  label.setAttribute('class','wk');
  label.setAttribute('x', p.x);
  label.setAttribute('y', BASE_Y + 22);
  label.textContent = p.w.tag || ('W' + p.w.week);

  g.appendChild(focusRing);
  g.appendChild(dot);
  g.appendChild(label);
  svg.appendChild(g);

  g.addEventListener('click', () => selectWeek(i));
  g.addEventListener('keydown', (e) => {
    if(e.key === 'Enter' || e.key === ' '){ e.preventDefault(); selectWeek(i); }
    if(e.key === 'ArrowRight'){ e.preventDefault(); selectWeek(Math.min(i+1, n-1)); document.querySelectorAll('.tab-point')[Math.min(i+1,n-1)].focus(); }
    if(e.key === 'ArrowLeft'){ e.preventDefault(); selectWeek(Math.max(i-1, 0)); document.querySelectorAll('.tab-point')[Math.max(i-1,0)].focus(); }
  });
});

function renderPanel(i){
  const w = WEEKS[i];
  const panel = document.getElementById('panel');

  const stats = [
    ['Swim', w.swim], ['Bike', w.bike], ['Run', w.run], ['Finish', w.finish]
  ].filter(([,v]) => v && v.trim() !== '');

  const statsHtml = stats.length ? `
    <div class="stat-row">
      ${stats.map(([k,v]) => `<div class="stat"><span class="k">${k}</span><span class="v">${v}</span></div>`).join('')}
    </div>` : '';

  const paragraphs = w.entry.trim().split(/\n\s*\n/).map(p => `<p>${p.trim().replace(/\n\s+/g,' ')}</p>`).join('');
  const weekLabel = w.tag === 'RACE' ? 'RACE DAY' : `WEEK ${String(w.week).padStart(2,'0')}`;
  const phasePill = w.phase ? `<span class="phase-tag">${w.phase}</span>` : '';

  panel.innerHTML = `
    <div class="panel-top">
      <span><span class="week-tag">${weekLabel}</span>${phasePill}</span>
      <span class="week-date">${w.dateRange}</span>
    </div>
    <div class="mood-headline">${MOOD_LABELS[w.mood] || ''}</div>
    ${statsHtml}
    <div class="entry">${paragraphs}</div>
    <div class="week-nav">
      <button class="nav-btn" id="prevBtn" ${i === 0 ? 'disabled' : ''} aria-label="Previous week">
        <span class="arrow">←</span> Prev
      </button>
      <span class="nav-hint">use ← → keys</span>
      <button class="nav-btn" id="nextBtn" ${i === WEEKS.length - 1 ? 'disabled' : ''} aria-label="Next week">
        Next <span class="arrow">→</span>
      </button>
    </div>
  `;

  document.getElementById('prevBtn').addEventListener('click', () => {
    if(activeIndex > 0) selectWeek(activeIndex - 1);
  });
  document.getElementById('nextBtn').addEventListener('click', () => {
    if(activeIndex < WEEKS.length - 1) selectWeek(activeIndex + 1);
  });
}

function selectWeek(i){
  activeIndex = i;
  document.querySelectorAll('.tab-point').forEach((el, idx) => {
    el.classList.toggle('active', idx === i);
    el.setAttribute('aria-selected', idx === i ? 'true' : 'false');
  });
  renderPanel(i);
}

selectWeek(activeIndex);

// Global left/right arrow navigation — works anywhere on the page,
// as long as focus isn't inside a text input (none here, but future-proof).
document.addEventListener('keydown', (e) => {
  const tag = document.activeElement.tagName;
  if(tag === 'INPUT' || tag === 'TEXTAREA') return;

  // The individual tab-point elements already handle their own
  // left/right for accessible roving focus, so skip when one of
  // those has focus to avoid double-stepping.
  if(document.activeElement.classList && document.activeElement.classList.contains('tab-point')) return;

  if(e.key === 'ArrowRight' && activeIndex < WEEKS.length - 1){
    e.preventDefault();
    selectWeek(activeIndex + 1);
  }
  if(e.key === 'ArrowLeft' && activeIndex > 0){
    e.preventDefault();
    selectWeek(activeIndex - 1);
  }
});
